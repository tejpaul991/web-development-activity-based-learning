function displayMovies(){
 const query=document.getElementById("search").value.toLowerCase();
 const genre=document.getElementById("genre").value;
 const results=movies.filter(m=>
  m.name.toLowerCase().includes(query)&&
  (genre==="All"||m.genre===genre)
 );
 document.getElementById("movieGrid").innerHTML=results.map(m=>`
 <div class="card">
 <img src="${m.img}" alt="${m.name}">
 <div><h3>${m.name}</h3>
 <p>${m.genre} | ${m.year}</p>
 <p class="rating">⭐ ${m.rating}/10</p>
 <p>${m.desc}</p>
 <p><b>Cast:</b> ${m.cast}</p>
 <button onclick="showDetails('${m.name}')">Read More</button>
 </div></div>`).join("")||"<p>No movies found.</p>";
}

function showDetails(name){
 const m=movies.find(movie=>movie.name===name);
 alert(m.name+" ("+m.year+")\nGenre: "+m.genre+"\nRating: "+m.rating+"/10\nCast: "+m.cast+"\n\n"+m.desc);
}

function addReview(){
 const name=document.getElementById("movieName").value.trim();
 const review=document.getElementById("reviewText").value.trim();
 if(!name||!review){alert("Enter a movie name and review.");return;}
 const p=document.createElement("p");
 p.textContent="🎬 "+name+": "+review;
 document.getElementById("reviewList").prepend(p);
 document.getElementById("movieName").value="";
 document.getElementById("reviewText").value="";
}
displayMovies();
